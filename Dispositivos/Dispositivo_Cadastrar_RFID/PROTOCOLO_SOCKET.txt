PROTOCOLO SOCKET.IO - CADASTRO RFID
=================================

1. REGISTRO DE HARDWARE
-----------------------
ESP32 -> Backend

evento:
registerHardware

{
  "deviceId": "RFID-AABBCCDDEEFF",
  "deviceType": "cadastro_rfid",
  "firmwareVersion": "1.0.0",
  "deviceToken": "opcional"
}


2. ATIVAÇÃO / VÍNCULO
---------------------
Backend -> ESP32

evento:
hardwareLinked

{
  "success": true,
  "deviceId": "RFID-AABBCCDDEEFF",
  "deviceToken": "123456789012345",
  "deviceType": "cadastro_rfid"
}


3. INICIAR CADASTRO
-------------------
Backend -> ESP32

evento:
iniciarCadastroRFID

{
  "artigoId": "MongoObjectId",
  "codigo": "ART001",
  "quantidade": 100,
  "jaCadastradas": 0,
  "sessionId": "opcional"
}

-----------
ESP32 -> Backend

evento:
epcCadastroRFID

{

  "artigoId": "MongoObjectId",
